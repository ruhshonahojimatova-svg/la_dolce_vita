function renderCart() {
  const count = [...cart.values()].reduce((a,b) => a+b, 0);
  document.getElementById("cartCount").textContent = count;
  const items = [...cart.entries()];
  document.getElementById("cartItems").innerHTML = items.length ? items.map(([id,qty]) => {
    const d = dishes.find(x => x.id === id);
    return `<div class="cart-item">
      <div><strong>${d.ru}</strong><small>${qty} × ${money(d.price)}</small></div>
      <div style="text-align:right"><strong>${money(d.price*qty)}</strong><br>
      <button class="remove-item" data-remove="${id}">Убрать</button></div>
    </div>`;
  }).join("") : '<div class="cart-empty">Пока ничего нет. Выберите любимое блюдо в меню.</div>';
  const total = items.reduce((sum,[id,qty]) => sum + dishes.find(d=>d.id===id).price*qty, 0);
  document.getElementById("cartTotal").textContent = money(total);
}
